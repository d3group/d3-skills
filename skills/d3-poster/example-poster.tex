%%% Motivation %%%
\headerbox{Motivation}{name=Motivation,column=0,row=0}{
  Periodic-review inventory control with fixed ordering cost is widely studied
  for the single-period case, but the \highlight{multi-period rolling-horizon}
  variant has remained data-poor. We apply weighted SAA end to end to 460 SKUs
  of an East-African pharmaceutical distributor.
}

%%% Problem -- formal setup, fills column 0 to the bottom
\headerbox{Problem}{name=Problem,column=0,below=Motivation,above=bottom}{
  At each period $t = 1,\dots,T$, decide $q_t \geq 0$ before observing demand
  $\xi_t$. Total cost:
  \begin{equation*}
    C(\vec{q},\vec{\xi},I_1) = \sum_{t=1}^{T}\bigl(c^o_t(q_t) + c^s_t(z_t,\xi_t)\bigr),
  \end{equation*}
  with $c^o_t(q) = K_t + u_t q$ for $q>0$ and newsvendor cost
  $c^s_t(z,\xi) = h_t(z-\xi)^+ + b_t(\xi-z)^+$. Given features $\vec{x}$:
  \begin{equation*}
    \min_{q_1,\dots,q_T \geq 0} \mathbb{E}[C(\vec{q},\vec{\xi},I_1) \mid \vec{x}].
  \end{equation*}

  \keypoint{Closed-form $\mathbb{E}[\cdot\mid\vec{x}]$ is unavailable; we
  estimate it with a weighted sample average trained on global data.}
}

%%% Approach -- top of column 1
\headerbox{Approach}{name=Approach,column=1,row=0}{
  \textbf{Weighted SAA.} Replace the conditional expectation with
  $\sum_{i=1}^{N} w^{i}(\vec{x})\,C(\vec{q},\vec{\xi}^i,I_1)$, where weights
  come from a globally-trained ML model applied locally per SKU.

  \vspace{0.4ex}
  \textbf{Rolling horizon.} Solve a $(\tau{+}1)$-period look-ahead at each $t$,
  apply only $\hat{q}_t$, roll forward.
