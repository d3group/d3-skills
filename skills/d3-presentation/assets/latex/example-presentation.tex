\onecol{One-column slides present content in a single flow}{
    This is a standard content slide with a single column layout.

    \begin{itemize}
        \item First bullet point with some explanation
        \item Second bullet point with more details
        \item Third point with \textbf{bold text} and \highlight{highlighted text}
    \end{itemize}

    \vspace{3mm}
    \navybox{Navy box for key messages}
    \hspace{3mm}
    \textbf{\textcolor{d3-orange}{Orange text for emphasis}}
}

\twocol{Two-column slides enable side-by-side comparison}{
    \columnheader{Left Column}

    Content in the left column:
    \begin{itemize}
        \item Point one
        \item Point two
        \item Point three
    \end{itemize}
}{
    \columnheader{Right Column}

    Content in the right column:
    \begin{itemize}
        \item Another point
        \item More content
        \item Final item
    \end{itemize}
}

\twocoltakeaway{Takeaway slides highlight the key insight for the audience}{
    \columnheader{Analysis}

    \begin{itemize}
        \item Data point one
        \item Analysis finding
        \item Supporting evidence
    \end{itemize}
}{
    \columnheader{Implications}

    \begin{itemize}
        \item Strategic recommendation
        \item Action item
        \item Expected outcome
    \end{itemize}

    \begin{codebox}

    \end{codebox}

    \vspace{3mm}
    \textbf{Error box} (red border):
    \begin{errorbox}
ERROR: File not found
    \end{errorbox}
}{

    \begin{terminalbox}
\$ python main.py\\
Processing data...\\

    \end{terminalbox}

    \vspace{3mm}
    Use \texttt{\textbackslash pyind} for indentation in code boxes.
}

\showsection{3}

\onecol{Two-phase process flow}{
    \vspace{4mm}
    \twochevron{1}{Phase 1:}{Current focus}{Phase 2:}{Next phase}

    \vspace{6mm}
    {\small
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textit{Current phase:}
            \begin{itemize}
                \item Description of current phase activities
                \item Key deliverables and milestones
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textit{Next phase:}
            \begin{itemize}
                \item Planned activities for the next phase
                \item Expected outcomes and dependencies
            \end{itemize}
        \end{column}
    \end{columns}
    }
}

\onecol{Three-phase process flow}{
    \vspace{4mm}
    \threechevron{2}{Phase 1:}{Discover}{Phase 2:}{Design}{Phase 3:}{Deliver}

    \vspace{6mm}
    {\small
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.32\textwidth}
            \textit{Discover:}
            \begin{itemize}
                \item Research
                \item Requirements
            \end{itemize}
        \end{column}
        \begin{column}{0.32\textwidth}
            \textit{Design:}
            \begin{itemize}
                \item Architecture
                \item Prototypes
            \end{itemize}
        \end{column}
        \begin{column}{0.32\textwidth}
            \textit{Deliver:}
            \begin{itemize}
                \item Build
                \item Deploy
            \end{itemize}
        \end{column}
    \end{columns}
    }
}

\onecol{Four-phase process flow}{
    \vspace{4mm}

    \vspace{6mm}
    {\small
    \begin{columns}[T,onlytextwidth]

            \textit{Plan:}
            \begin{itemize}

            \end{itemize}
        \end{column}
        \begin{column}{0.24\textwidth}

        \node[d3box] (process) at (3.5,0) {Process\\Step};
        \node[d3box] (analyze) at (7,0) {Analyze\\Results};
        \node[d3project] (output) at (10.5,0) {Final\\Output};

        \draw[d3arrow] (input) -- (process);
        \draw[d3arrow] (process) -- (analyze);
        \draw[d3arrow] (analyze) -- (output);

        \node[below, font=\scriptsize, text=d3-gray] at (input.south) {d3box};
        \node[below, font=\scriptsize, text=d3-gray] at (process.south) {d3box};
        \node[below, font=\scriptsize, text=d3-gray] at (analyze.south) {d3box};
        \node[below, font=\scriptsize, text=d3-orange] at (output.south) {d3project};
    \end{tikzpicture}
    \end{center}
